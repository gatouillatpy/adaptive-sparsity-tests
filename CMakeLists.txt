cmake_minimum_required(VERSION 3.18)
project(adaptive_sparsity LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenMP REQUIRED)
option(AVEC_CUDA "Build CUDA comparison" OFF)
add_executable(bench bench.cpp essai.cpp)
target_link_libraries(bench PRIVATE OpenMP::OpenMP_CXX ${CMAKE_DL_LIBS})
if(MSVC)
  target_compile_options(bench PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/O2;/arch:AVX2>)
else()
  target_compile_options(bench PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3;-march=native>)
endif()
if(AVEC_CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(bench PRIVATE gpu.cpp)
  configure_file(gpu.cu gpu.cu COPYONLY)
  target_compile_definitions(bench PRIVATE AVEC_CUDA)
  target_link_libraries(bench PRIVATE CUDA::cudart CUDA::cublas CUDA::cuda_driver CUDA::nvrtc)
endif()
enable_testing()
add_test(NAME cpu COMMAND bench --self-test)
if(AVEC_CUDA)
  add_test(NAME gpu_fp32 COMMAND bench --self-test --backend gpu --precision fp32)
  add_test(NAME gpu_fp16 COMMAND bench --self-test --backend gpu --precision fp16)
endif()
